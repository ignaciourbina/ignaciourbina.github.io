% .panel -- white, 6px radius, the CSS box-shadow rendered as a blurred drop
% shadow. The content sits inside at --pad on every side, in a fixed-height
% top-aligned box so nothing can spill past the panel edge unnoticed.
\newcommand{\panel}[5]{% x, y from page top, width, height (in), content
    \node[anchor=north west, inner sep=0pt, fill=paper,
          rounded corners=10.8pt,
          minimum width=#3in, minimum height=#4in,
          blur shadow={shadow blur steps=10, shadow xshift=0pt,
                       shadow yshift=-3.6pt, shadow opacity=30,
                       shadow blur radius=18pt}]
        at (#1in, -#2in) {};
    \node[anchor=north west, inner sep=0pt]
        at (\fpeval{#1+\PadIn}in, -\fpeval{#2+\PadIn}in) {%
        \begin{minipage}[t][\fpeval{#4-2*\PadIn}in][t]{\fpeval{#3-2*\PadIn}in}%
            \setlength{\parindent}{0pt}\setlength{\parskip}{0pt}%
            % Browsers do not hyphenate body prose, so neither does this.
            \raggedright\frenchspacing
            \hyphenpenalty=10000 \exhyphenpenalty=10000
            \color{ink}#5%
        \end{minipage}};%
}

% The header band: navy-deep, full bleed, centred title over byline over a
% letter-spaced uppercase affiliation.
\newcommand{\posterheader}[5]{%
    \fill[navydeep] (0,0) rectangle (\PosterW in, -\HeaderH in);
    % #4 -- QR to the author's site, left of the title, on a white tile so it
    % scans against the navy band. Vertically centred in the header.
    \node[anchor=west, inner sep=7pt, fill=white, rounded corners=5pt]
        at (\SideMargin in, -\fpeval{\HeaderH/2}in)
        {\qrcode[height=1.25in, level=M, tight]{#4}};
    % #5 -- what the QR is for, to its right. The minipage is deliberately
    % narrow (2.2in) so the line wraps and stops well short of the title,
    % which begins around 8in.
    \node[anchor=west, inner sep=0pt] at (2.20in, -\fpeval{\HeaderH/2}in) {%
        \begin{minipage}{2.2in}
            \raggedright\SZ{11}{14.5}\color{byline}#5\par
        \end{minipage}};
    \node[anchor=north, inner sep=0pt] at (\fpeval{\PosterW/2}in, -0.45in) {%
        \begin{minipage}{44in}\centering
            {\fTitle\SZ{27}{31}\color{white}#1\par}
            \vspace{\PX{4}}
            {\SZ{18.5}{22}\color{byline}#2\par}
            \vspace{\PX{2}}
            {\fAffil\SZ{13.6}{16.3}\color{affil}\MakeUppercase{#3}\par}
        \end{minipage}};%
}

% The funding line, centred in the bottom margin under the canvas. The canvas
% ends at \CanvasY + \CanvasH = 26.4in, leaving 0.6in of navy before the page
% edge; the line sits in that band, in the same faint blue as the affiliation.
\newcommand{\posterfooter}[1]{%
    \node[anchor=north, inner sep=0pt]
        at (\fpeval{\PosterW/2}in, -\fpeval{\CanvasY+\CanvasH+0.16}in) {%
        \begin{minipage}{44in}\centering
            {\SZ{10.5}{14}\color{white}\raggedright\centering#1\par}
        \end{minipage}};%
}
